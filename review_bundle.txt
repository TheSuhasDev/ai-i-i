
===== ml-service\app\decision.py =====

"""
Decision engine (Day 8): rule-based, no ML. Pure functions plus a small CLI (python -m app.decision).

Safety stock   SS = z * sqrt(LT * sigma_d^2 + d^2 * sigma_LT^2)
  z = 1.65 (95% service), sigma_d = std of daily demand, d = mean daily demand, LT = lead time (days),
  sigma_LT = std of supplier delay (days). With sigma_LT = 0 this is the Day 2 / Day 7 label definition.

Smart reorder
  daily    = forecast demand over lead time / LT
  eff_LT   = LT + average supplier delay          (a habitually late supplier must be covered for the delay)
  demand   = daily * eff_LT
  buffer   = SS * (1 + 0.5 * (1 - reliability / 100))    (1.0x for a perfect supplier, 1.5x at score 0)
  target   = demand + buffer                      (this is the reorder point)
  order    = max(0, ceil(target - (stock + on_order)))

ABC: annual value share; A = products needed to reach 70%, B = up to 90%, C = rest.
XYZ: coefficient of variation of weekly demand (52 weeks); X < 0.25, Y < 0.35, Z otherwise.

Supplier reliability (0-100) = 100 * (0.40 on-time + 0.25 delay + 0.20 delay-variance + 0.15 quality),
each part scaled to 0-1. Risk score = 100 - reliability.
"""
import math

import numpy as np
import pandas as pd

from app.preprocessing import DATA_DIR
from app.risk import load_series

Z_SERVICE = 1.65
RELIABILITY_BUFFER = 0.5

ABC_A_CUT, ABC_B_CUT = 0.70, 0.90   # calibrated against scenario_labels.abc_actual (exact match)
XYZ_X_CUT, XYZ_Y_CUT = 0.25, 0.35   # calibrated against scenario_labels.xyz_actual (90% agreement)

PRIOR_N = 5               # pseudo-orders of prior when scoring a supplier
ONTIME_PRIOR = 0.85
QUALITY_PRIOR = 0.05
W_ONTIME, W_DELAY, W_VARIANCE, W_QUALITY = 0.40, 0.25, 0.20, 0.15
DELAY_CAP_DAYS = 7.0      # an average delay of 7+ days scores 0 on the delay part
STD_CAP_DAYS = 5.0        # a delay std of 5+ days scores 0 on the variance part
QUALITY_CAP = 0.20        # 20%+ of POs with quality issues scores 0 on the quality part

PRICE_COLS = ["unit_price", "selling_price", "price", "unit_cost", "cost_price", "cost"]
QUALITY_COLS = ["quality_issue", "quality_issues", "has_quality_issue", "quality_issue_flag", "defective", "defect_flag"]


# ---------------------------------------------------------------- safety stock and reorder

def safety_stock(sigma_daily, lead_time_days, mean_daily=0.0, lt_std_days=0.0, z=Z_SERVICE):
    if lead_time_days <= 0:
        raise ValueError("lead_time_days must be positive")
    var = lead_time_days * float(sigma_daily) ** 2 + (float(mean_daily) * float(lt_std_days)) ** 2
    return float(z * math.sqrt(var))


def smart_reorder(forecast_ltd, lead_time_days, sigma_daily, stock, on_order=0.0,
                  reliability=100.0, avg_delay_days=0.0, lt_std_days=0.0, z=Z_SERVICE):
    """forecast_ltd = forecast demand over the nominal lead time. reliability = supplier score 0-100."""
    if lead_time_days <= 0:
        raise ValueError("lead_time_days must be positive")
    reliability = min(max(float(reliability), 0.0), 100.0)
    daily = max(float(forecast_ltd), 0.0) / lead_time_days
    eff_lt = lead_time_days + max(float(avg_delay_days), 0.0)
    demand = daily * eff_lt
    ss = safety_stock(sigma_daily, lead_time_days, daily, lt_std_days, z)
    buffer = ss * (1 + RELIABILITY_BUFFER * (1 - reliability / 100))
    target = demand + buffer
    position = float(stock) + float(on_order)
    qty = max(0, math.ceil(target - position - 1e-9))
    return {
        "order_qty": int(qty),
        "needs_reorder": qty > 0,
        "reorder_point": round(target, 1),
        "demand_over_lead_time": round(demand, 1),
        "safety_stock": round(buffer, 1),
        "effective_lead_time_days": round(eff_lt, 1),
        "inventory_position": round(position, 1),
        "days_of_cover": round(float(stock) / daily, 1) if daily > 0 else None,
    }


# ---------------------------------------------------------------- ABC-XYZ

def abc_classify(values):
    """values: annual value per product (Series). Returns a Series of 'A'/'B'/'C' with the same index."""
    v = pd.Series(values, dtype=float).clip(lower=0)
    total = v.sum()
    if total <= 0:
        return pd.Series("C", index=v.index)
    ordered = v.sort_values(ascending=False)
    share_before = (ordered.cumsum() - ordered) / total
    cls = np.select([share_before < ABC_A_CUT, share_before < ABC_B_CUT], ["A", "B"], "C")
    return pd.Series(cls, index=ordered.index).reindex(v.index)


def xyz_classify(cv):
    c = pd.Series(cv, dtype=float)
    cls = np.select([c < XYZ_X_CUT, c < XYZ_Y_CUT], ["X", "Y"], "Z")
    return pd.Series(cls, index=c.index)


# ---------------------------------------------------------------- supplier reliability

def _flag(s):
    if s.dtype == object:
        return s.astype(str).str.strip().str.lower().isin(["true", "yes", "y", "1", "1.0"]).astype(float)
    return (pd.to_numeric(s, errors="coerce").fillna(0) > 0).astype(float)


def reliability_from_metrics(on_time, avg_delay, delay_std, quality_rate):
    s = (W_ONTIME * min(max(on_time, 0.0), 1.0)
         + W_DELAY * (1 - min(avg_delay / DELAY_CAP_DAYS, 1))
         + W_VARIANCE * (1 - min(delay_std / STD_CAP_DAYS, 1))
         + W_QUALITY * (1 - min(quality_rate / QUALITY_CAP, 1)))
    return round(100 * s, 1)


def supplier_scores(pos):
    """pos needs: supplier_id, status, delay_days, quality_issue. Only delivered POs count."""
    d = pos[pos["status"] == "Delivered"].copy()
    d["delay_days"] = pd.to_numeric(d["delay_days"], errors="coerce").fillna(0.0)
    d["late"] = (d["delay_days"] > 0).astype(float)
    d["delay_pos"] = d["delay_days"].clip(lower=0)
    d["issue"] = _flag(d["quality_issue"])
    g = d.groupby("supplier_id")
    out = pd.DataFrame({
        "delivered_pos": g.size(),
        "late_pos": g["late"].sum(),
        "avg_delay_days": g["delay_pos"].mean(),
        "delay_std_days": g["delay_days"].std(ddof=1).fillna(0.0),
        "issue_pos": g["issue"].sum(),
    })
    n = out["delivered_pos"]
    out["on_time_rate"] = ((n - out["late_pos"]) + ONTIME_PRIOR * PRIOR_N) / (n + PRIOR_N)
    out["quality_issue_rate"] = (out["issue_pos"] + QUALITY_PRIOR * PRIOR_N) / (n + PRIOR_N)
    out["reliability_score"] = [
        reliability_from_metrics(r.on_time_rate, r.avg_delay_days, r.delay_std_days, r.quality_issue_rate)
        for r in out.itertuples()]
    out["risk_score"] = (100 - out["reliability_score"]).round(1)
    out["risk_level"] = np.select([out["risk_score"] < 20, out["risk_score"] < 40], ["Low", "Medium"], "High")
    return out.drop(columns=["late_pos", "issue_pos"]).round(3)


NEUTRAL_SUPPLIER = {"reliability_score": reliability_from_metrics(ONTIME_PRIOR, 0.0, 0.0, QUALITY_PRIOR),
                    "avg_delay_days": 0.0, "delay_std_days": 0.0}


# ---------------------------------------------------------------- data loading (CLI and Day 9)

def _need(df, names, what, fname):
    for n in names:
        if n in df.columns:
            return n
    raise ValueError(f"No {what} column in {fname}. Looked for {names}; the file has {list(df.columns)}. "
                     f"Add the right name to the list at the top of decision.py.")


def load_decision_inputs():
    products = pd.read_csv(DATA_DIR / "products.csv")
    pos = pd.read_csv(DATA_DIR / "purchase_orders.csv")
    price_col = _need(products, PRICE_COLS, "unit price", "products.csv")
    qcol = _need(pos, QUALITY_COLS, "quality issue", "purchase_orders.csv")
    pos = pos.rename(columns={qcol: "quality_issue"})
    if "supplier_id" not in products.columns:
        ids = _need(products, ["supplier_ids", "supplier"], "supplier", "products.csv")
        # a product can list several suppliers; the first one is treated as its primary supplier
        products["supplier_id"] = products[ids].astype(str).str.findall(r"[^\s,;|\[\]'\"]+").str[0]
    if "supplier_id" not in pos.columns:
        pos = pos.merge(products[["product_id", "supplier_id"]], on="product_id", how="left")
    return products, pos, price_col


def product_stats(series, products, price_col):
    meta = products.set_index("product_id")
    rows = []
    for pid, s in series.items():
        f = s["frame"]
        u = f["units_sold"].astype(float)
        weekly = pd.Series(u.to_numpy(), index=pd.DatetimeIndex(f["date"])).resample("W-SUN").sum().tail(52)
        mean_w = weekly.mean()
        rows.append({
            "product_id": pid, "name": s["name"], "lead_time_days": s["lt"],
            "stock": float(f["stock_on_hand"].iloc[-1]),
            "mean_daily_28": float(u.tail(28).mean()),
            "sigma_daily_60": float(u.tail(60).std(ddof=1)),
            "weekly_cv": float(weekly.std(ddof=1) / mean_w) if mean_w > 0 else float("inf"),
            "annual_value": float(u.tail(365).sum() * float(meta.loc[pid, price_col])),
            "supplier_id": str(meta.loc[pid, "supplier_id"]),
        })
    df = pd.DataFrame(rows)
    df["abc"] = abc_classify(df["annual_value"].set_axis(df["product_id"])).to_numpy()
    df["xyz"] = xyz_classify(df["weekly_cv"]).to_numpy()
    df["abc_xyz"] = df["abc"] + df["xyz"]
    return df


def main():
    products, pos, price_col = load_decision_inputs()
    stats = product_stats(load_series(), products, price_col)
    sup = supplier_scores(pos)
    print(f"Products whose supplier has no PO history: {int((~stats['supplier_id'].isin(sup.index)).sum())} of {len(stats)}")
    print("Supplier reliability:")
    print(sup.sort_values("reliability_score").to_string())

    print(f"\nABC-XYZ matrix (products):")
    print(pd.crosstab(stats["abc"], stats["xyz"]).to_string())
    print(f"A products: {int((stats['abc'] == 'A').sum())} | weekly CV quartiles: "
          f"{stats['weekly_cv'].replace(np.inf, np.nan).quantile([.25, .5, .75]).round(2).tolist()}")

    sup_map = sup.to_dict("index")
    rows = []
    for r in stats.to_dict("records"):
        sc = sup_map.get(r["supplier_id"], NEUTRAL_SUPPLIER)
        res = smart_reorder(r["mean_daily_28"] * r["lead_time_days"], r["lead_time_days"], r["sigma_daily_60"],
                            r["stock"], reliability=sc["reliability_score"],
                            avg_delay_days=sc["avg_delay_days"], lt_std_days=sc["delay_std_days"])
        rows.append({"product_id": r["product_id"], "name": r["name"], "abc_xyz": r["abc_xyz"],
                     "stock": r["stock"], "reorder_point": res["reorder_point"], "order_qty": res["order_qty"],
                     "supplier_score": sc["reliability_score"]})
    out = pd.DataFrame(rows)
    print(f"\nProducts needing a reorder now: {int(out['order_qty'].gt(0).sum())} of {len(out)} "
          f"(demand stand-in: last 28 days x lead time, no open POs counted)")
    print(out.sort_values("order_qty", ascending=False).head(10).to_string(index=False))

    scen = pd.read_csv(DATA_DIR / "scenario_labels.csv")[["product_id", "stock_scenario"]]
    chk = out.merge(scen, on="product_id")
    print("\nShare needing reorder by planted stock scenario (expect reorder high, over/well low):")
    print(chk.groupby("stock_scenario")["order_qty"].agg(products="size", needing=lambda s: int((s > 0).sum())).to_string())


if __name__ == "__main__":
    main()

===== ml-service\app\insights.py =====

"""
One combined result per product (Day 9): forecast, risk, trend, reorder, health and alerts.
Run `python -m app.insights` to validate against scenario_labels.csv (validation only, never used by the code).
"""
from datetime import date, timedelta
from functools import lru_cache

import numpy as np
import pandas as pd

from app.decision import NEUTRAL_SUPPLIER, load_decision_inputs, product_stats, smart_reorder, supplier_scores
from app.forecasting import forecast_product
from app.health import OVERSTOCK_DAYS, PRIORITY_RANK, alert_priority, health_score
from app.preprocessing import DATA_DIR
from app.risk import load_series, risk_for_product
from app.trend import trend_for_product


@lru_cache(maxsize=1)
def _tables():
    products, pos, price_col = load_decision_inputs()
    stats = product_stats(load_series(), products, price_col).set_index("product_id").to_dict("index")
    return stats, supplier_scores(pos).to_dict("index")


@lru_cache(maxsize=1)
def _open_pos():
    """Open (not yet delivered) purchase orders per product: total quantity and earliest expected delivery."""
    pos = pd.read_csv(DATA_DIR / "purchase_orders.csv", parse_dates=["expected_delivery_date"])
    o = pos[pos["status"].isin(["Pending", "In Transit"])].copy()
    if o.empty:
        return {}
    received = o["qty_received"].fillna(0) if "qty_received" in o else 0
    o["qty_remaining"] = (o["qty_ordered"] - received).clip(lower=0)
    g = o.groupby("product_id").agg(qty=("qty_remaining", "sum"), date=("expected_delivery_date", "min"))
    return {str(pid): {"qty": float(r.qty), "date": r.date.date()} for pid, r in g.iterrows()}


def _finite(x, nd=2):
    x = float(x)
    return round(x, nd) if np.isfinite(x) else None


def _alerts(prio, margin, cover, days_deliv, reorder, sup_id, sc):
    out = []
    if prio != "Low":
        if margin < 0:
            msg = (f"Stock covers about {cover:.1f} days but replenishment is expected in {days_deliv:.1f} days "
                   f"(short by {-margin:.1f} days).")
        else:
            msg = f"Stock covers {cover:.1f} days against {days_deliv:.1f} days to replenishment (margin {margin:.1f} days)."
        out.append({"type": "stockout_before_delivery" if margin < 0 else "low_cover_margin", "priority": prio,
                    "message": msg})
    if reorder["order_qty"] > 0:
        out.append({"type": "reorder_now", "priority": "High" if prio in ("Critical", "High") else "Medium",
                    "message": f"Order {reorder['order_qty']} units (reorder point {reorder['reorder_point']}, "
                               f"inventory position {reorder['inventory_position']})."})
    if cover is not None and cover > OVERSTOCK_DAYS:
        out.append({"type": "overstock", "priority": "Low",
                    "message": f"{cover:.0f} days of cover; consider pausing reorders."})
    if sc.get("risk_level") == "High" and prio != "Low":
        out.append({"type": "supplier_risk", "priority": "Medium",
                    "message": f"Supplier {sup_id} is high risk (reliability {sc['reliability_score']}, "
                               f"average delay {sc['avg_delay_days']:.1f} days)."})
    return sorted(out, key=lambda a: PRIORITY_RANK[a["priority"]])


def insights_for_product(product_id, stock=None, on_order=None, promo_days=None, price=None,
                         expected_delivery_date=None):
    """stock / on_order optionally override the CSV snapshot (Express sends live values from MongoDB)."""
    pid = str(product_id)
    stats, sup = _tables()
    row = stats.get(pid)
    if row is None:
        raise KeyError(pid)

    fc = forecast_product(pid, promo_days, price)
    rk = risk_for_product(pid, stock)
    try:
        tr = trend_for_product(pid)
    except ValueError:
        tr = None

    sup_id = str(row["supplier_id"])
    sc = sup.get(sup_id, NEUTRAL_SUPPLIER)
    lt = int(row["lead_time_days"])
    as_of = date.fromisoformat(rk["as_of"])
    cur = float(rk["stock"])
    po = _open_pos().get(pid)
    qty_in = float(on_order) if on_order is not None else (po["qty"] if po else 0.0)

    reorder = smart_reorder(fc["lead_time_demand"], lt, float(row["sigma_daily_60"]), cur, qty_in,
                            sc["reliability_score"], sc["avg_delay_days"], sc["delay_std_days"])

    daily = float(np.mean([w["forecast"] for w in fc["weeks"]])) / 7
    cover = min(cur / daily, 365.0) if daily > 0.05 else None
    delay = float(sc["avg_delay_days"])
    expected_date = expected_delivery_date or (po["date"] if po else None)
    if isinstance(expected_date, str):
        expected_date = date.fromisoformat(expected_date)
    if qty_in > 0 and expected_date:
        days_deliv = max((expected_date - as_of).days, 0) + delay
        basis = "open purchase order"
    else:
        days_deliv = lt + delay
        basis = "new order placed today (lead time + average supplier delay)"
    margin = None if cover is None else round(cover - days_deliv, 1)
    prio = alert_priority(margin)

    health = health_score(rk["signals"]["stock_ratio"], cover, rk["risk_pct"], sc["reliability_score"],
                          float(row["weekly_cv"]))

    return {
        "product_id": pid,
        "name": row["name"],
        "as_of": rk["as_of"],
        "inputs": {"stock": cur, "on_order": qty_in, "lead_time_days": lt,
                   "stock_source": "override" if stock is not None else "csv snapshot"},
        "priority": prio,
        "alerts": _alerts(prio, margin, cover, days_deliv, reorder, sup_id, sc),
        "health": health,
        "stockout": {
            "days_of_cover": None if cover is None else round(cover, 1),
            "expected_stockout_date": None if cover is None else (as_of + timedelta(days=cover)).isoformat(),
            "days_to_delivery": round(days_deliv, 1),
            "expected_delivery_date": (as_of + timedelta(days=round(days_deliv))).isoformat(),
            "delivery_basis": basis,
            "margin_days": margin,
        },
        "forecast": fc,
        "risk": rk,
        "trend": tr,
        "reorder": {**reorder, "supplier_id": sup_id, "supplier_reliability": sc["reliability_score"]},
        "classification": {"abc": str(row["abc"]), "xyz": str(row["xyz"]), "abc_xyz": str(row["abc_xyz"]),
                           "weekly_cv": _finite(row["weekly_cv"])},
        "supplier": {"id": sup_id, "reliability_score": sc["reliability_score"],
                     "risk_level": sc.get("risk_level", "Low"), "avg_delay_days": _finite(delay, 1)},
    }


def all_insights():
    out = []
    for pid in load_series():
        try:
            out.append(insights_for_product(pid))
        except ValueError:
            continue
    return out

def supplier_table():
    """Supplier reliability/risk table for the Supplier Risk page. NaN/inf become null (JSON-safe)."""
    _, sup = _tables()
    out = []
    for sid, row in sup.items():
        clean = {}
        for k, v in row.items():
            if isinstance(v, (int, float, np.integer, np.floating)) and not isinstance(v, bool):
                v = float(v)
                clean[k] = v if np.isfinite(v) else None
            else:
                clean[k] = None if v is None or (not isinstance(v, str) and pd.isna(v)) else v
        out.append({"supplier_id": str(sid), **clean})
    return sorted(out, key=lambda r: r.get("reliability_score") if r.get("reliability_score") is not None else 0)

def main():
    res = all_insights()
    df = pd.DataFrame([{
        "product_id": r["product_id"], "name": r["name"], "health": r["health"]["score"],
        "status": r["health"]["status"], "priority": r["priority"], "margin": r["stockout"]["margin_days"],
        "cover": r["stockout"]["days_of_cover"], "to_delivery": r["stockout"]["days_to_delivery"],
        "order_qty": r["reorder"]["order_qty"], "alerts": len(r["alerts"])} for r in res])
    lab = pd.read_csv(DATA_DIR / "scenario_labels.csv")[
        ["product_id", "expected_status", "expected_alert", "health_score", "cover_margin_days", "stock_scenario"]]
    m = df.merge(lab, on="product_id")

    print("Health status: mine (rows) vs provisional label (columns)")
    print(pd.crosstab(m["status"], m["expected_status"]).to_string())
    print(f"exact agreement {(m['status'] == m['expected_status']).mean() * 100:.0f}% | "
          f"correlation of health score with label {m['health'].corr(m['health_score']):.2f}")

    print("\nAlert priority: mine (rows) vs provisional label (columns)")
    print(pd.crosstab(m["priority"], m["expected_alert"]).to_string())
    gap = (m["priority"].map(PRIORITY_RANK) - m["expected_alert"].map(PRIORITY_RANK)).abs()
    print(f"exact agreement {(gap == 0).mean() * 100:.0f}% | within one level {(gap <= 1).mean() * 100:.0f}%")

    bad = m[(m["status"] != m["expected_status"]) | (gap > 0)]
    print("\nDisagreements:")
    print(bad[["product_id", "name", "stock_scenario", "health", "health_score", "status", "expected_status",
               "margin", "cover_margin_days", "priority", "expected_alert"]].round(1).to_string(index=False)
          if len(bad) else "  none")

    print("\nMost urgent products:")
    top = df.assign(rank=df["priority"].map(PRIORITY_RANK)).sort_values(["rank", "health"]).head(10)
    print(top[["product_id", "name", "priority", "health", "status", "cover", "to_delivery", "order_qty"]]
          .to_string(index=False))


if __name__ == "__main__":
    main()


===== ml-service\app\routes_insights.py =====

from datetime import date
from typing import List, Optional

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel, Field

from app.insights import all_insights, insights_for_product, supplier_table

router = APIRouter()

NOT_TRAINED = "Model not trained. Run: python -m app.train_forecast and python -m app.train_risk"


class InsightRequest(BaseModel):
    product_id: str
    stock: Optional[float] = Field(default=None, ge=0, description="Current on-hand stock. Default: CSV snapshot.")
    on_order: Optional[float] = Field(default=None, ge=0, description="Quantity on open POs. Default: CSV open POs.")
    promo_days: Optional[List[int]] = Field(default=None, description="Planned promo days (0-7) for the next 3 weeks.")
    price: Optional[float] = Field(default=None, gt=0)
    expected_delivery_date: Optional[date] = None


@router.post("/insights")
def insights(req: InsightRequest):
    if req.promo_days is not None and (len(req.promo_days) > 3 or any(not 0 <= d <= 7 for d in req.promo_days)):
        raise HTTPException(status_code=422, detail="promo_days: up to 3 values, each between 0 and 7")
    try:
        return insights_for_product(req.product_id, req.stock, req.on_order, req.promo_days, req.price,
                                    req.expected_delivery_date)
    except FileNotFoundError:
        raise HTTPException(status_code=503, detail=NOT_TRAINED)
    except KeyError:
        raise HTTPException(status_code=404, detail=f"Unknown product_id {req.product_id}")
    except ValueError as e:
        raise HTTPException(status_code=422, detail=str(e))


@router.get("/insights/all")
def insights_all():
    try:
        return all_insights()
    except FileNotFoundError:
        raise HTTPException(status_code=503, detail=NOT_TRAINED)

@router.get("/supplier-risk")
def supplier_risk():
    return supplier_table()


===== ml-service\app\simulator.py =====

"""
What-if simulator (Day 12).

Recalculates one product's demand, stockout date, stockout risk, reorder quantity, health and alert priority
for changed inputs and compares them with the current situation ("before") on the same code path.

Inputs
  demand_pct        demand change applied to the forecast and, for the risk model, to the last 28 days of sales
  extra_delay_days  additional supplier delay: stretches the effective lead time and shifts every arriving order
  stock             scenario stock on hand (default: current stock)
  order_qty         a new order placed today; it arrives after lead time + average supplier delay + extra delay

Notes
  * "before" reproduces /insights exactly (checked by `python -m app.simulator`).
  * The risk model does not see orders in transit (Day 7), so order_qty changes the projected stockout date,
    the units short and the suggested additional order, not the risk %.
  * Daily demand = mean of the 3-week forecast, held constant over the 30-day projection (no weekday effect).
  * Dates follow the /insights convention: expected stockout = as_of + whole days of cover.
"""
from datetime import date, timedelta

import numpy as np
import pandas as pd

from app.decision import smart_reorder
from app.forecasting import forecast_product, lead_time_demand
from app.health import PRIORITY_RANK, alert_priority, health_score
from app.insights import NEUTRAL_SUPPLIER, _open_pos, _tables
from app.risk import RFEATS, apply_calibration, load_bundle, load_series, product_features, risk_level

HORIZON = 30
WINDOW = 28


# ---------------------------------------------------------------- pure helpers

def scale_recent_units(frame, factor, window=WINDOW):
    """Copy of the daily frame with the last `window` days of sales multiplied by `factor`."""
    f = frame.copy()
    f["units_sold"] = f["units_sold"].astype(float)
    if factor != 1.0:
        idx = f.index[-window:]
        f.loc[idx, "units_sold"] = f.loc[idx, "units_sold"] * factor
    return f


def project_stock(stock, daily, arrivals, days=HORIZON):
    """
    Day-by-day stock with constant daily demand. `arrivals` is a list of (days_from_now, qty);
    an order arriving after d days is added at the start of day d+1, which matches the margin
    used for alerts (cover - days to delivery).
    """
    s = float(stock)
    levels, short_total, first_short = [], 0.0, None
    for t in range(1, days + 1):
        s += sum(q for d, q in arrivals if max(int(round(d)), 0) == t - 1)
        if s >= daily:
            s -= daily
        else:
            short_total += daily - s
            s = 0.0
            if first_short is None:
                first_short = t
        levels.append(round(s, 1))
    return {"levels": levels, "first_shortfall_day": first_short, "units_short": round(short_total, 1)}


# ---------------------------------------------------------------- risk under a scenario

def _risk_row(frame, lt, stock, factor, extra_delay):
    f = scale_recent_units(frame, factor)
    f.loc[f.index[-1], "stock_on_hand"] = float(stock)
    row = product_features(f, lt + extra_delay).iloc[-1]   # extra delay = longer effective lead time
    if row[RFEATS].isna().any():
        raise ValueError("Not enough history (need 60+ days) to simulate this product")
    bundle = load_bundle()
    X = pd.DataFrame([row[RFEATS].astype(float).to_dict()])
    raw = bundle["model"].predict_proba(X)[:, 1]
    p = float(apply_calibration(bundle["calibrator"], raw)[0])
    return row, p, bundle["threshold"]


def _state(c, stock, on_order, new_order, factor, extra):
    lt, delay = c["lt"], c["delay"]
    lt_eff = lt + extra
    daily = c["daily"] * factor
    cover = min(stock / daily, 365.0) if daily > 0.05 else None

    arrivals = []
    if on_order > 0:
        if c["po"] is not None:
            when = max((c["po"]["date"] - c["as_of"]).days, 0) + delay + extra
        else:
            when = lt + delay + extra
        arrivals.append((when, on_order))
    if new_order > 0:
        arrivals.append((lt + delay + extra, new_order))
    days_deliv = min(w for w, _ in arrivals) if arrivals else lt + delay + extra
    margin = None if cover is None else round(cover - days_deliv, 1)
    prio = alert_priority(margin)

    proj = project_stock(stock, daily, arrivals)
    first = proj["first_shortfall_day"]

    row, p, thr = _risk_row(c["frame"], lt, stock, factor, extra)
    pct = round(p * 100, 1)

    ltd = round(lead_time_demand([w * factor for w in c["weekly"]], lt_eff), 1)
    sc = c["sc"]
    reorder = smart_reorder(ltd, lt_eff, c["sigma"] * factor, stock, on_order + new_order,
                            sc["reliability_score"], sc["avg_delay_days"], sc["delay_std_days"])
    health = health_score(round(float(row["stock_ratio"]), 2), cover, pct, sc["reliability_score"], c["weekly_cv"])

    return {
        "stock": round(float(stock), 1),
        "on_order": round(float(on_order + new_order), 1),
        "daily_demand": round(float(daily), 2),
        "lead_time_days": int(lt_eff),
        "lead_time_demand": ltd,
        "days_of_cover": None if cover is None else round(float(cover), 1),
        "days_to_delivery": round(float(days_deliv), 1),
        "margin_days": margin,
        "expected_stockout_date": None if first is None else (c["as_of"] + timedelta(days=first - 1)).isoformat(),
        "first_shortfall_day": first,
        "units_short_30d": proj["units_short"],
        "risk_pct": pct,
        "risk_level": risk_level(pct),
        "flagged": bool(p >= thr),
        "priority": prio,
        "health": health,
        "reorder": reorder,
        "projected_stock": proj["levels"],
    }


def _summary(b, a, neutral):
    if neutral:
        return "No scenario changes applied: showing the current situation."
    parts = [f"Alert priority moves from {b['priority']} to {a['priority']}." if a["priority"] != b["priority"]
             else f"Alert priority stays {a['priority']}."]
    parts.append(f"Stockout risk goes from {b['risk_pct']:.0f}% to {a['risk_pct']:.0f}%.")
    if a["health"]["status"] != b["health"]["status"]:
        parts.append(f"Health moves from {b['health']['status']} to {a['health']['status']}.")
    if a["expected_stockout_date"] != b["expected_stockout_date"]:
        was = b["expected_stockout_date"] or "none within 30 days"
        now = a["expected_stockout_date"] or "none within 30 days"
        parts.append(f"Expected stockout moves from {was} to {now}.")
    parts.append(f"Suggested additional order: {b['reorder']['order_qty']} -> {a['reorder']['order_qty']} units.")
    return " ".join(parts)


def simulate(product_id, demand_pct=0.0, extra_delay_days=0, stock=None, order_qty=0.0,
             base_stock=None, base_on_order=None, base_expected_delivery_date=None):
    pid = str(product_id)
    stats, sup = _tables()
    row = stats.get(pid)
    series = load_series().get(pid)
    if row is None or series is None:
        raise KeyError(pid)
    factor = 1.0 + float(demand_pct) / 100.0
    if factor <= 0:
        raise ValueError("demand_pct must be greater than -100")
    extra = int(extra_delay_days)

    fc = forecast_product(pid)
    frame = series["frame"]
    sup_id = str(row["supplier_id"])
    sc = sup.get(sup_id, NEUTRAL_SUPPLIER)
    po = _open_pos().get(pid)
    expected_date = base_expected_delivery_date or (po["date"] if po else None)
    if isinstance(expected_date, str):
        expected_date = date.fromisoformat(expected_date)
    if expected_date:
        po = {**(po or {}), "date": expected_date}
    weekly = [w["forecast"] for w in fc["weeks"]]

    cur = float(frame["stock_on_hand"].iloc[-1]) if base_stock is None else float(base_stock)
    qty_in = float(base_on_order) if base_on_order is not None else (po["qty"] if po else 0.0)
    ctx = {
        "frame": frame, "lt": int(row["lead_time_days"]), "delay": float(sc["avg_delay_days"]), "sc": sc,
        "sigma": float(row["sigma_daily_60"]), "weekly_cv": float(row["weekly_cv"]), "weekly": weekly,
        "daily": float(np.mean(weekly)) / 7, "po": po, "as_of": frame["date"].iloc[-1].date(),
    }

    scen_stock = cur if stock is None else float(stock)
    before = _state(ctx, cur, qty_in, 0.0, 1.0, 0)
    after = _state(ctx, scen_stock, qty_in, float(order_qty), factor, extra)
    neutral = demand_pct == 0 and extra == 0 and stock is None and float(order_qty) == 0

    return {
        "product_id": pid,
        "name": str(row["name"]),
        "supplier_id": sup_id,
        "as_of": ctx["as_of"].isoformat(),
        "scenario": {"demand_pct": float(demand_pct), "extra_delay_days": extra,
                     "stock": scen_stock, "order_qty": float(order_qty)},
        "before": before,
        "after": after,
        "flags": {
            "priority_changed": before["priority"] != after["priority"],
            "priority_worse": PRIORITY_RANK[after["priority"]] < PRIORITY_RANK[before["priority"]],
            "risk_level_changed": before["risk_level"] != after["risk_level"],
            "health_status_changed": before["health"]["status"] != after["health"]["status"],
        },
        "summary": _summary(before, after, neutral),
        "assumptions": [
            "Demand change applies to the forecast and to the last 28 days of sales seen by the risk model.",
            "Extra delay lengthens the effective lead time and delays every arriving order.",
            "The risk model ignores orders in transit, so order quantity does not change the risk %; "
            "it changes the stockout date, units short and the suggested additional order.",
            "The 30-day projection holds daily demand constant at the forecast mean.",
        ],
    }

def simulate_all(demand_pct=0.0, extra_delay_days=0, live=None, product_ids=None):
    """
    Same scenario applied to every product (or to `product_ids`), as compact rows, worst affected first.
    `live` maps product_id -> {"stock": .., "on_order": ..} (current values sent by Express).
    """
    live = live or {}
    ids = [str(p) for p in product_ids] if product_ids else list(load_series())
    rows = []
    for pid in ids:
        base = live.get(pid) or {}
        try:
            s = simulate(pid, demand_pct, extra_delay_days, None, 0.0, base.get("stock"), base.get("on_order"),
                         base.get("expected_delivery_date"))
        except ValueError:
            continue
        b, a = s["before"], s["after"]
        rows.append({
            "product_id": pid, "name": s["name"],
            "priority_before": b["priority"], "priority_after": a["priority"],
            "priority_worse": s["flags"]["priority_worse"],
            "risk_before": b["risk_pct"], "risk_after": a["risk_pct"],
            "days_of_cover_before": b["days_of_cover"], "days_of_cover_after": a["days_of_cover"],
            "order_qty_before": b["reorder"]["order_qty"], "order_qty_after": a["reorder"]["order_qty"],
            "expected_stockout_after": a["expected_stockout_date"],
        })
    rows.sort(key=lambda r: (not r["priority_worse"], PRIORITY_RANK[r["priority_after"]], -r["risk_after"]))
    n = max(len(rows), 1)
    return {
        "scenario": {"demand_pct": float(demand_pct), "extra_delay_days": int(extra_delay_days)},
        "summary": {
            "n_products": len(rows),
            "priority_worse": int(sum(r["priority_worse"] for r in rows)),
            "critical_before": int(sum(r["priority_before"] == "Critical" for r in rows)),
            "critical_after": int(sum(r["priority_after"] == "Critical" for r in rows)),
            "avg_risk_before": round(sum(r["risk_before"] for r in rows) / n, 1),
            "avg_risk_after": round(sum(r["risk_after"] for r in rows) / n, 1),
            "reorder_units_before": round(sum(r["order_qty_before"] for r in rows)),
            "reorder_units_after": round(sum(r["order_qty_after"] for r in rows)),
        },
        "products": rows,
    }


# ---------------------------------------------------------------- validation

SCENARIOS = {
    "demand +10%": {"demand_pct": 10},
    "demand +20%": {"demand_pct": 20},
    "delay +3d": {"extra_delay_days": 3},
    "demand +20% & delay +3d": {"demand_pct": 20, "extra_delay_days": 3},
}


def main():
    from app.insights import insights_for_product
    from app.preprocessing import DATA_DIR

    base, mismatches = [], []
    for pid in load_series():
        try:
            s = simulate(pid)
            ins = insights_for_product(pid)
        except ValueError:
            continue
        b, diffs = s["before"], []
        if b["priority"] != ins["priority"]:
            diffs.append("priority")
        if abs(b["risk_pct"] - ins["risk"]["risk_pct"]) > 0.05:
            diffs.append("risk")
        if abs(b["health"]["score"] - ins["health"]["score"]) > 0.05:
            diffs.append("health")
        if b["reorder"]["order_qty"] != ins["reorder"]["order_qty"]:
            diffs.append("order_qty")
        if diffs:
            mismatches.append((pid, ",".join(diffs)))
        base.append((pid, s))
    print(f"Baseline vs /insights: {len(base) - len(mismatches)} of {len(base)} products identical "
          f"(priority, risk, health, order qty)")
    if mismatches:
        print("  MISMATCHES (the simulator is not on the same code path):", mismatches)

    recs = []
    for pid, s in base:
        rec = {"product_id": pid, "name": s["name"], "risk": s["before"]["risk_pct"], "prio": s["before"]["priority"]}
        for label, kw in SCENARIOS.items():
            rec[label] = simulate(pid, **kw)["after"]
        recs.append(rec)

    print(f"\nEffect of each scenario on all {len(recs)} products:")
    for label in SCENARIOS:
        changed = sum(r[label]["priority"] != r["prio"] for r in recs)
        worse = sum(PRIORITY_RANK[r[label]["priority"]] < PRIORITY_RANK[r["prio"]] for r in recs)
        band = sum(r[label]["risk_level"] != risk_level(r["risk"]) for r in recs)
        fell = sum(r[label]["risk_pct"] < r["risk"] - 0.5 for r in recs)
        print(f"  {label:26s} priority changed {changed:2d} (worse {worse:2d}) | risk band changed {band:2d} "
              f"| risk fell by more than 0.5 pt: {fell}")

    lab = pd.read_csv(DATA_DIR / "scenario_labels.csv")[["product_id", "stock_scenario"]]
    near = set(lab[lab["stock_scenario"] == "near"]["product_id"])

    def cell(x):
        return f"{x['risk_pct']:.0f}% {x['priority']}"

    lines = [{"product": f"{r['product_id']} {r['name']}", "now": f"{r['risk']:.0f}% {r['prio']}",
              "demand +20%": cell(r["demand +20%"]), "delay +3d": cell(r["delay +3d"]),
              "both": cell(r["demand +20% & delay +3d"])} for r in recs if r["product_id"] in near]
    print(f"\nPlanted near-threshold products ({len(lines)}): risk % and alert priority")
    print(pd.DataFrame(lines).to_string(index=False))
    flipped = sum(r["demand +20% & delay +3d"]["priority"] != r["prio"] for r in recs if r["product_id"] in near)
    print(f"Priority changes under 'demand +20% & delay +3d': {flipped} of {len(lines)} near-threshold products")


if __name__ == "__main__":
    main()

import PurchaseOrder from "../models/PurchaseOrder.js";

const OPEN_STATUSES = ["Pending", "In Transit"];

export function summarizeOpenPurchaseOrders(orders) {
  return orders.filter((order) => OPEN_STATUSES.includes(order.status)).reduce(
    (summary, order) => {
      const remaining = Math.max(Number(order.qtyOrdered || 0) - Number(order.qtyReceived || 0), 0);
      summary.onOrder += remaining;
      if (remaining > 0 && order.expectedDeliveryDate &&
          (!summary.expectedDeliveryDate || order.expectedDeliveryDate < summary.expectedDeliveryDate)) {
        summary.expectedDeliveryDate = order.expectedDeliveryDate;
      }
      return summary;
    },
    { onOrder: 0, expectedDeliveryDate: null }
  );
}

export async function getOpenPurchaseOrderSummary(productId) {
  const orders = await PurchaseOrder.find({ product: productId, status: { $in: OPEN_STATUSES } })
    .select("qtyOrdered qtyReceived expectedDeliveryDate")
    .lean();
  return summarizeOpenPurchaseOrders(orders);
}

import AppError from "../utils/AppError.js";

const DEFAULT_TIMEOUT_MS = 20000;
const PASS_THROUGH = new Set([404, 422, 503]);

async function call(path, { method = "GET", body, timeoutMs = DEFAULT_TIMEOUT_MS } = {}) {
  const base = process.env.ML_SERVICE_URL || "http://localhost:8000";
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  try {
    const res = await fetch(`${base}${path}`, {
      method,
      headers: {
        "Content-Type": "application/json",
        "X-Service-Token": process.env.ML_SERVICE_TOKEN || "",
      },
      body: body ? JSON.stringify(body) : undefined,
      signal: controller.signal,
    });
    const data = await res.json().catch(() => null);
    if (!res.ok) {
      const detail = typeof data?.detail === "string" ? data.detail : `ML service returned ${res.status}`;
      throw new AppError(detail, PASS_THROUGH.has(res.status) ? res.status : 502);
    }
    return data;
  } catch (err) {
    if (err instanceof AppError) throw err;
    throw new AppError(err.name === "AbortError" ? "ML service timed out" : "ML service is unreachable", 503);
  } finally {
    clearTimeout(timer);
  }
}

const RANK = { Critical: 0, High: 1, Medium: 2, Low: 3 };
const DAY = 86400000;

export async function computeAndStore(product) {
  const inv = await Inventory.findOne({ product: product._id });
  const purchaseOrders = await getOpenPurchaseOrderSummary(product._id);
  const body = { product_id: product.code };
  if (inv) {
    body.stock = inv.currentStock;
  }
  body.on_order = purchaseOrders.onOrder;
  if (purchaseOrders.expectedDeliveryDate) {
    body.expected_delivery_date = purchaseOrders.expectedDeliveryDate.toISOString().slice(0, 10);
  }
  const data = await mlService.insights(body);
  return Insight.findOneAndUpdate(
    { product: product._id },
    {
      productCode: product.code,
      asOf: data.as_of,
      priority: data.priority,
      healthScore: data.health.score,
      healthStatus: data.health.status,
      riskPct: data.risk.risk_pct,
      orderQty: data.reorder.order_qty,
      daysOfCover: data.stockout.days_of_cover,
      marginDays: data.stockout.margin_days,
      data,
      computedAt: new Date(),
    },
    { upsert: true, returnDocument: "after", setDefaultsOnInsert: true }
  );
}

// GET /api/insights/:code  -> full combined result for one product (recomputed and stored)
export async function getInsight(req, res) {
  const product = await Product.findOne({ code: req.params.code.toUpperCase() });
  if (!product) throw new AppError(`Unknown product ${req.params.code}`, 404);
  const doc = await computeAndStore(product);
  res.json({ success: true, data: { ...doc.data, computedAt: doc.computedAt } });
}

// GET /api/insights?priority=&status=&full=1  -> stored results, most urgent first.
// Default returns light summaries; full=1 includes the whole payload (alerts, reorder, ...).
export async function listInsights(req, res) {
  const filter = {};
  if (req.query.priority) filter.priority = req.query.priority;
  if (req.query.status) filter.healthStatus = req.query.status;
  let query = Insight.find(filter);
  if (req.query.full !== "1") {
    query = query.select(
      "productCode priority healthScore healthStatus riskPct orderQty daysOfCover marginDays computedAt data.name data.classification data.trend.pattern data.trend.trend data.trend.yoy_change_pct data.trend.monthly_index data.trend.weekly_series"
    );
  }
  const docs = await query.lean();
  docs.sort((a, b) => RANK[a.priority] - RANK[b.priority] || a.healthScore - b.healthScore);
  res.json({ success: true, count: docs.length, data: docs });
}

// GET /api/insights/:code/history?weeks=52  -> complete Monday-Sunday weeks of units sold
export async function getHistory(req, res) {
  const product = await Product.findOne({ code: req.params.code.toUpperCase() });
  if (!product) throw new AppError(`Unknown product ${req.params.code}`, 404);
  const weeks = Math.min(Math.max(parseInt(req.query.weeks) || 52, 4), 156);

  const last = await Sale.findOne({ product: product._id }).sort("-date").select("date").lean();
  if (!last) return res.json({ success: true, data: { code: product.code, weeks: [] } });

  const since = new Date(last.date.getTime() - (weeks + 1) * 7 * DAY);
  const rows = await Sale.aggregate([
    { $match: { product: product._id, date: { $gt: since } } },
    {
      $group: {
        _id: { y: { $isoWeekYear: "$date" }, w: { $isoWeek: "$date" } },
        weekStart: { $min: "$date" },
        units: { $sum: "$unitsSold" },
        days: { $sum: 1 },
      },
    },
    { $sort: { weekStart: 1 } },
  ]);
  const data = rows
    .filter((r) => r.days === 7) // complete weeks only, like the ML service
    .slice(-weeks)
    .map((r) => ({ weekStart: r.weekStart.toISOString().slice(0, 10), units: r.units }));
  res.json({ success: true, data: { code: product.code, weeks: data } });
}

// POST /api/insights/refresh  -> recompute every active product (admin, manager)
export async function refreshAll(req, res) {
  const products = await Product.find({ isActive: true });
  const result = { total: products.length, ok: 0, failed: [] };
  for (let i = 0; i < products.length; i += 5) {
    const batch = products.slice(i, i + 5);
    const settled = await Promise.allSettled(batch.map(computeAndStore));
    settled.forEach((s, j) => {
      if (s.status === "fulfilled") result.ok += 1;
      else {
        console.error(`Insight refresh failed for ${batch[j].code}:`, s.reason);
        result.failed.push({ code: batch[j].code, error: "INSIGHT_COMPUTE_FAILED" });
      }
    });
  }
  res.json({ success: true, data: result });
}

// GET /api/insights/:code/explain  -> why the stockout risk is what it is (uses live MongoDB stock)
export async function getExplanation(req, res) {
  const product = await Product.findOne({ code: req.params.code.toUpperCase() });
  if (!product) throw new AppError(`Unknown product ${req.params.code}`, 404);
  const inv = await Inventory.findOne({ product: product._id });
  const body = { product_id: product.code };
  if (inv) body.stock = inv.currentStock;
  res.json({ success: true, data: await mlService.explain(body) });
}

import PurchaseOrder from "../models/PurchaseOrder.js";
import { makeCrud } from "../controllers/crudFactory.js";
import { crudRouter } from "./crudRouter.js";
import { poCreate, poUpdate } from "../validators/schemas.js";

const DAY = 86400000;

// delayDays = actual - expected delivery (negative = early). Matches the Day 2 definition.
function withDelay(data, existing = {}) {
  const out = { ...data };
  const actual = out.actualDeliveryDate !== undefined ? out.actualDeliveryDate : existing.actualDeliveryDate;
  const expected = out.expectedDeliveryDate ?? existing.expectedDeliveryDate;
  if (actual && expected) {
    if (out.delayDays === undefined) {
      out.delayDays = Math.round((new Date(actual) - new Date(expected)) / DAY);
    }
    if (!out.status) out.status = "Delivered";
  }
  return out;
}

const crud = makeCrud({
  Model: PurchaseOrder,
  label: "Purchase order",
  populate: [
    { path: "product", select: "code name" },
    { path: "supplier", select: "code name" },
  ],
  searchFields: ["poNumber"],
  filterFields: ["product", "supplier", "status", "qualityIssue"],
  defaultSort: "-orderDate",
  beforeCreate: (data) => withDelay(data),
  beforeUpdate: (data, existing) => withDelay(data, existing),
});

export default crudRouter(crud, { createSchema: poCreate, updateSchema: poUpdate });
